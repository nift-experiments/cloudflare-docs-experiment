<p>Send emails with the <code>EMAIL</code> binding and handle incoming emails with the <code>email()</code> handler in <code>src/index.ts</code>:</p>
<pre><code class="language-ts">interface Env {
	EMAIL: SendEmail;
}

export default {
	// Handle HTTP requests (Email Sending)
	async fetch(request, env, ctx): Promise&lt;Response&gt; {
		// Send a welcome email
		await env.EMAIL.send({
			to: &quot;user@example.com&quot;,
			from: &quot;welcome@yourdomain.com&quot;,
			subject: &quot;Welcome to our service!&quot;,
			html: &quot;&lt;h1&gt;Welcome!&lt;/h1&gt;&lt;p&gt;Thanks for signing up.&lt;/p&gt;&quot;,
			text: &quot;Welcome! Thanks for signing up.&quot;,
		});

		return new Response(&quot;Email sent successfully&quot;);
	},

	// Handle incoming emails (Email Routing)
	async email(message, env, ctx): Promise&lt;void&gt; {
		// Forward to support team
		if (message.to.includes(&quot;support@yourdomain.com&quot;)) {
			await message.forward(&quot;team@yourdomain.com&quot;);
		}

		// Send auto-reply
		await env.EMAIL.send({
			to: message.from,
			from: &quot;noreply@yourdomain.com&quot;,
			subject: &quot;We received your message&quot;,
			html: &quot;&lt;h1&gt;Thank you!&lt;/h1&gt;&lt;p&gt;We'll get back to you soon.&lt;/p&gt;&quot;,
		});
	},
} satisfies ExportedHandler&lt;Env&gt;;
</code></pre>
<p>Add the bindings to your Wrangler configuration file:</p>
<pre><code class="language-jsonc">{
	&quot;$schema&quot;: &quot;node_modules/wrangler/config-schema.json&quot;,
	&quot;name&quot;: &quot;&lt;ENTER_WORKER_NAME&gt;&quot;,
	&quot;main&quot;: &quot;src/index.ts&quot;,
	&quot;compatibility_date&quot;: &quot;$today&quot;,

	// Email sending
	&quot;send_email&quot;: [
		{
			&quot;name&quot;: &quot;EMAIL&quot;
		}
	],

	// Email routing
	&quot;email&quot;: [
		{
			&quot;name&quot;: &quot;EMAIL_HANDLER&quot;
		}
	]
}
</code></pre>
