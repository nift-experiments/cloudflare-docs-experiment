

<details class="nb-details"><summary>Request volume overview</summary><div class="nb-details-body">

<p>Get a high-level view of daily request volume over a date range.</p>
<pre><code class="language-graphql">query DailyRequestVolume(
  $accountTag: String!
  $startDate: Date!
  $endDate: Date!
) {
  viewer {
    accounts(filter: { accountTag: $accountTag }) {
      privacyProxyRequestMetricsAdaptiveGroups(
        filter: {
          date_geq: $startDate
          date_leq: $endDate
        }
        limit: 10000
        orderBy: [date_ASC]
      ) {
        count
        dimensions {
          date
        }
      }
    }
  }
}
</code></pre>
<pre><code class="language-json">{
  &quot;accountTag&quot;: &quot;&lt;YOUR_ACCOUNT_TAG&gt;&quot;,
  &quot;startDate&quot;: &quot;2026-04-04&quot;,
  &quot;endDate&quot;: &quot;2026-04-06&quot;
}
</code></pre>

<details class="nb-details"><summary>Error breakdown by status code and proxy status</summary><div class="nb-details-body">

<p>Identify which HTTP status codes and proxy-level errors are occurring to pinpoint the source of failures.</p>
<pre><code class="language-graphql">query ErrorBreakdown(
  $accountTag: String!
  $start: Time!
  $end: Time!
) {
  viewer {
    accounts(filter: { accountTag: $accountTag }) {
      privacyProxyRequestMetricsAdaptiveGroups(
        filter: {
          datetimeFifteenMinutes_geq: $start
          datetimeFifteenMinutes_leq: $end
          statusCode_geq: 400
        }
        limit: 10000
        orderBy: [datetimeFifteenMinutes_ASC]
      ) {
        count
        dimensions {
          datetimeFifteenMinutes
          statusCode
          proxyStatus
        }
      }
    }
  }
}
</code></pre>
<pre><code class="language-json">{
  &quot;accountTag&quot;: &quot;&lt;YOUR_ACCOUNT_TAG&gt;&quot;,
  &quot;start&quot;: &quot;2026-04-04T00:00:00Z&quot;,
  &quot;end&quot;: &quot;2026-04-06T23:59:59Z&quot;
}
</code></pre>
