

<details class="nb-details"><summary>Auth volume by method</summary><div class="nb-details-body">

<p>Track daily authentication volume per method to understand adoption and spot anomalies.</p>
<pre><code class="language-graphql">query AuthVolumeByMethod(
  $accountTag: String!
  $startDate: Date!
  $endDate: Date!
) {
  viewer {
    accounts(filter: { accountTag: $accountTag }) {
      privacyProxyAuthMetricsAdaptiveGroups(
        filter: {
          date_geq: $startDate
          date_leq: $endDate
        }
        limit: 10000
        orderBy: [date_ASC]
      ) {
        count
        dimensions {
          date
          authMethod
        }
      }
    }
  }
}
</code></pre>

  &quot;startDate&quot;: &quot;2026-04-04&quot;,

<details class="nb-details"><summary>Auth failure spike detection</summary><div class="nb-details-body">

<p>Detect surges in authentication failures and identify which auth method is failing.</p>
<pre><code class="language-graphql">query AuthFailureSpike(
  $accountTag: String!
  $start: Time!
  $end: Time!
) {
  viewer {
    accounts(filter: { accountTag: $accountTag }) {
      privacyProxyAuthMetricsAdaptiveGroups(
        filter: {
          datetimeMinute_geq: $start
          datetimeMinute_leq: $end
          authResult: &quot;failure&quot;
        }
        limit: 10000
        orderBy: [datetimeFiveMinutes_ASC]
      ) {
        count
        dimensions {
          datetimeFiveMinutes
          authMethod
        }
      }
    }
  }
}
</code></pre>
<pre><code class="language-json">{
  &quot;accountTag&quot;: &quot;&lt;YOUR_ACCOUNT_TAG&gt;&quot;,
  &quot;start&quot;: &quot;2026-04-04T10:00:00Z&quot;,
  &quot;end&quot;: &quot;2026-04-06T14:00:00Z&quot;
}
</code></pre>
