	},
});

	new PutObjectCommand({

		Key: &quot;user-upload.png&quot;,
		ContentType: &quot;image/png&quot;,
	}),
	{ expiresIn: 3600 }, // Valid for 1 hour
);

console.log(presignedUrl);
