<h3 id="body-11746-build-a-custom-controlbar">Build a custom controlbar</h3>

Replace `RtkControlbar` with individual components in a `View`:

```tsx
import React from "react";
import { View, StyleSheet } from "react-native";
import {
	RtkMicToggle,
	RtkCameraToggle,
	RtkLeaveButton,
	RtkMoreToggle,
	RtkScreenShareToggle,
} from "@cloudflare/realtimekit-react-native-ui";

function CustomControlbar({ meeting }) {
	return (
		<View style={styles.controlbar}>
			<RtkMicToggle meeting={meeting} variant="horizontal" />
			<RtkCameraToggle meeting={meeting} variant="horizontal" />
			<RtkScreenShareToggle meeting={meeting} variant="horizontal" />
			<RtkMoreToggle meeting={meeting} variant="horizontal" />
			<RtkLeaveButton />
		</View>
	);
}

const styles = StyleSheet.create({
	controlbar: {
		flexDirection: "row",
		justifyContent: "space-evenly",
		alignItems: "center",
		backgroundColor: "#1A1A1A",
		paddingVertical: 8,
		paddingHorizontal: 16,
	},
});
```

<h3 id="body-11746-use-the-custom-controlbar-in-your-meeting-screen">Use the custom controlbar in your meeting screen</h3>

In your `MeetingScreens` component from [Build Your Own UI](/realtime/realtimekit/ui-kit/build-your-own-ui/), replace:

```tsx
<RtkControlbar meeting={meeting} />
```

with:

```tsx
<CustomControlbar meeting={meeting} />
```

<h3 id="body-11746-add-custom-buttons-using-rtkcontrolbarbutton-1">Add custom buttons using RtkControlbarButton</h3>

Use `RtkControlbarButton` to create buttons that match the RealtimeKit design:

```tsx
import { RtkControlbarButton } from "@cloudflare/realtimekit-react-native-ui";

function CustomControlbar({ meeting }) {
	return (
		<View style={styles.controlbar}>
			<RtkMicToggle meeting={meeting} variant="horizontal" />
			<RtkCameraToggle meeting={meeting} variant="horizontal" />
			<RtkControlbarButton
				label="Raise Hand"
				icon={raiseHandSvg}
				onClick={() => {
					// Custom raise hand logic
				}}
				variant="horizontal"
			/>
			<RtkLeaveButton />
		</View>
	);
}
```

<aside class="nb-aside note">


<p>Wrap your component tree in <code>RtkUIProvider</code>. All UI Kit components read design tokens and state from <code>RtkUIContext</code>.</p>


</aside>
