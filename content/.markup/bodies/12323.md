
Initialize the RealtimeKit SDK using the `useRealtimeKitClient` hook.

<pre><code class="language-js">import React from &#x27;react&#x27;;
import { View, Text } from &#x27;react-native&#x27;;
import { useRealtimeKitClient, RealtimeKitProvider } from &#x27;@cloudflare/realtimekit-react-native&#x27;;

export default function App() {
  const [meeting, initMeeting] = useRealtimeKitClient();
  React.useEffect(() =&gt; {
    const init = async () =&gt; {
      const meetingOptions = {
        audio: true,
        video: true,
      };
      await initMeeting({
        authToken: &#x27;YourAuthToken&#x27;,
        defaults: meetingOptions,
      });
    };
    init();
    // next - if (meeting) meeting.joinRoom();
  }, []);

  if (meeting) {
    return (
      &lt;RealtimeKitProvider value={meeting}&gt;
        {/* Render your components here */}
      &lt;/RealtimeKitProvider&gt;
    );
  } else {
    return (
      &lt;View&gt;
        &lt;Text&gt;Loading...&lt;/Text&gt;
      &lt;View&gt;
    )
  }
}
</code></pre>


Use the [Add participant API](/api/resources/realtime_kit/subresources/meetings/methods/add_participant/) to fetch the `authToken`.
