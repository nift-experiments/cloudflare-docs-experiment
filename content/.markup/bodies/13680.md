const result = await sandbox.exec('npm run build');

// With streaming
await sandbox.exec('npm install', {
  stream: true,
  onOutput: (stream, data) => console.log(`[${stream}] ${data}`)
});

// Pass input via stdin (no shell injection risks)

});
