<p>Configure the <a href="https://registry.terraform.io/providers/cloudflare/cloudflare/latest/docs/resources/ruleset"><code>cloudflare_ruleset</code></a> resource:</p>
<pre><code class="language-tf">resource &quot;cloudflare_ruleset&quot; &quot;account_level_network_ddos_config&quot; {
  account_id  = var.cloudflare_account_id
  name        = &quot;Network-layer DDoS Attack Protection entry point ruleset&quot;
  description = &quot;&quot;
  kind        = &quot;root&quot;
  phase       = &quot;ddos_l4&quot;

  rules = [{
    ref         = &quot;override_l7_ddos_ruleset_dst_ip&quot;
    description = &quot;Override the HTTP DDoS Attack Protection managed ruleset&quot;
    expression  = &quot;ip.dst in { 192.0.2.0/24 }&quot;
    action      = &quot;execute&quot;
    action_parameters = {
      # Cloudflare L3/4 DDoS Attack Protection Ruleset
      id = &quot;3b64149bfa6e4220bbbc2bd6db589552&quot;
      overrides = {
        rules = [{
          # Rule: Generic high-volume UDP traffic flows.
          id                = &quot;599dab0942ff4898ac1b7797e954e98b&quot;
          sensitivity_level = &quot;low&quot;
        }]
      }
    }
  }]
}
</code></pre>
