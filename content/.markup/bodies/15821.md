				return error;
			}
		}

		const finalResponse = await env.AI.run(

			{
				messages: [
					{

						content: JSON.stringify(selected_tool),
					},
					{
						role: &quot;tool&quot;,
						content: JSON.stringify(res),
					},
				],
				tools: [
					{
						name: &quot;getGithubUser&quot;,
						description:
							&quot;Provides publicly available information about someone with a GitHub account.&quot;,
						parameters: {
							type: &quot;object&quot;,
