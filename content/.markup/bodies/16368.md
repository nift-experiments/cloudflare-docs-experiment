

```ts
import { Hono } from "hono";
import { html } from "hono/html";

const app = new Hono();

app.get("*", (c) => {
	const doc = html`<!DOCTYPE html>
		<body>
			<h1>Hello World</h1>
			<p>This markup was generated by a Cloudflare Worker with Hono.</p>
		</body>`;

	return c.html(doc);
});

export default app;
```
