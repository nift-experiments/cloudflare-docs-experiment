

1. Create a custom server entrypoint file:

<div class="nb-type-script-example">

<pre><code class="language-ts">import handler from &quot;@tanstack/react-start/server-entry&quot;;

// Export Durable Objects as named exports
export { MyDurableObject } from &quot;./my-durable-object&quot;;

export default {
	fetch: handler.fetch,

	// Handle Queue messages
	async queue(batch, env, ctx) {
		for (const message of batch.messages) {
			console.log(&quot;Processing message:&quot;, message.body);
			message.ack();
		}
	},

	// Handle Cron Triggers
	async scheduled(event, env, ctx) {
		console.log(&quot;Cron triggered:&quot;, event.cron);
	},
};
</code></pre>

</div>
