<h3 class="nb-aside-title" id="body-17079-service-workers-are-deprecated">Service Workers are deprecated</h3>


<p>Service Workers are deprecated, but still supported. We recommend using <a href="/workers/reference/migrate-to-module-workers/">Module Workers</a> instead. New features may not be supported for Service Workers.</p>


</aside>

```js
addEventListener("fetch", (event) => {
	event.respondWith(fetchAndStream(event.request));
});

async function fetchAndStream(request) {
	// Fetch from origin server.
	const response = await fetch(request);

	const { readable, writable } = new TransformStream({
		transform(chunk, controller) {
			controller.enqueue(modifyChunkSomehow(chunk));
		},
	});

	// Start pumping the body. NOTE: No await!
	response.body.pipeTo(writable);

	// ... and deliver our Response while that’s running.
	return new Response(readable, response);
}
```
