

<pre><code class="language-ts">import { DynamicWorkerExecutor } from &quot;@cloudflare/codemode&quot;;
import { codeMcpServer } from &quot;@cloudflare/codemode/mcp&quot;;
import { McpServer } from &quot;@modelcontextprotocol/sdk/server/mcp.js&quot;;
import { createLegacyMcpHandler } from &quot;agents/mcp&quot;;
import { z } from &quot;zod&quot;;

function createOrderServer() {
	const server = new McpServer({
		name: &quot;orders&quot;,
		version: &quot;1.0.0&quot;,
	});

	server.registerTool(
		&quot;get_order&quot;,
		{
			description: &quot;Get an order by ID&quot;,
			inputSchema: {
				orderId: z.string().describe(&quot;Order ID&quot;),
			},
		},
		async ({ orderId }) =&gt; ({
			structuredContent: {
				id: orderId,
				status: &quot;processing&quot;,
			},
			content: [
				{
					type: &quot;text&quot;,
					text: JSON.stringify({ id: orderId, status: &quot;processing&quot; }),
				},
			],
		}),
	);

	return server;
}

export default {
	async fetch(request, env, ctx): Promise&lt;Response&gt; {
		const upstream = createOrderServer();
		const executor = new DynamicWorkerExecutor({ loader: env.LOADER });
		const server = await codeMcpServer({
			server: upstream,
			executor,
		});

		return createLegacyMcpHandler(server, { route: &quot;/mcp&quot; })(
			request,
			env,
			ctx,
		);
	},
} satisfies ExportedHandler&lt;Env&gt;;
