import { useAgent } from "agents/react";
import { useState } from "react";

function Dashboard() {
	const [tools, setTools] = useState([]);
	const [servers, setServers] = useState({});

	const agent = useAgent({
		agent: "MyAgent",
		onMcpUpdate: (mcpState) => {
			setTools(mcpState.tools);
			setServers(mcpState.servers);
		},
	});

	return (
<div>
			<h2 id="body-2270-connected-servers">Connected Servers</h2>
			{Object.entries(servers).map(([id, server]) => (
<div key={id}>
					{server.name}: {server.state}
</div>
			))}

			<h2 id="body-2270-available-tools-tools-length">Available Tools ({tools.length})</h2>
			{tools.map((tool) => (
<div key={`${tool.serverId}-${tool.name}`}>{tool.name}</div>
			))}
</div>
	);
}
