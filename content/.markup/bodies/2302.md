			timestamp: Date.now().toString(),
		};
	}, []);

	const agent = useAgent({
		agent: "my-agent",
		query: asyncQuery,
	});

}

function App() {
	return (
		<Suspense fallback={<div>Authenticating...</div>}>

		</Suspense>
	);
}
