<pre><code class="language-js">const privateKeyStr = `-----BEGIN RSA PRIVATE KEY-----
....
-----END RSA PRIVATE KEY-----`;

const crypto = require(&quot;crypto&quot;);
const privateKey = crypto.createPrivateKey(privateKeyStr);

const fs = require(&quot;fs&quot;);
const zlib = require(&quot;zlib&quot;);
const readline = require(&quot;readline&quot;);

async function importAESGCMKey(keyBuffer) {
	try {
		// Ensure the key length is valid for AES
		if ([128, 192, 256].includes(256)) {
			return await crypto.webcrypto.subtle.importKey(
				&quot;raw&quot;,
				keyBuffer,
				{
					name: &quot;AES-GCM&quot;,
					length: 256,
				},
				true, // Whether the key is extractable (true in this case to allow for export later if needed)
				[&quot;encrypt&quot;, &quot;decrypt&quot;], // Use for encryption and decryption
			);
		} else {
			throw new Error(&quot;Invalid AES key length. Must be 128, 12, or 256 bits.&quot;);
		}
	} catch (error) {
		console.error(&quot;Failed to import key:&quot;, error);
		throw error;
	}
}

async function decryptData(encryptedData, aesKey, iv) {
	const decryptedData = await crypto.subtle.decrypt(
		{ name: &quot;AES-GCM&quot;, iv: iv },
		aesKey,
		encryptedData,
	);
	return new TextDecoder().decode(decryptedData);
}

async function decryptBase64(privateKey, data) {
	if (data.key === undefined) {
		return data;
	}

	const aesKeyBuf = crypto.privateDecrypt(
		{
			key: privateKey,
			oaepHash: &quot;SHA256&quot;,
		},
		Buffer.from(data.key, &quot;base64&quot;),
	);
	const aesKey = await importAESGCMKey(aesKeyBuf);

	const decryptedData = await decryptData(
		Buffer.from(data.data, &quot;base64&quot;),
		aesKey,
		Buffer.from(data.iv, &quot;base64&quot;),
	);

	return decryptedData.toString();
}

async function run() {
	let lineReader = readline.createInterface({
		input: fs.createReadStream(&quot;my_log.log.gz&quot;).pipe(zlib.createGunzip()),
	});

	lineReader.on(&quot;line&quot;, async (line) =&gt; {
		line = JSON.parse(line);

		const { Metadata, RequestBody, ResponseBody, ...remaining } = line;

		console.log({
			...remaining,
			Metadata: await decryptBase64(privateKey, Metadata),
			RequestBody: await decryptBase64(privateKey, RequestBody),
			ResponseBody: await decryptBase64(privateKey, ResponseBody),
		});
		console.log(&quot;--&quot;);
	});
}

run();

<li>Decrypt the encrypted log file using the private key.</li>

<p>Assuming that the logs were encrypted with the public key (for example <code>public_key.pem</code>), you can use the private key (<code>private_key.pem</code>) to decrypt the log file.</p>
<p>For example, if the encrypted logs are in a file named <code>encrypted_logs.bin</code>, you can decrypt it like this:</p>
<pre><code class="language-bash">openssl rsautl -decrypt -inkey private_key.pem -in encrypted_logs.bin -out decrypted_logs.txt
</code></pre>
<ul>
<li><code>-decrypt</code> tells OpenSSL that we want to decrypt the file.</li>
<li><code>-inkey private_key.pem</code> specifies the private key that will be used to decrypt the logs.</li>
<li><code>-in encrypted_logs.bin</code> is the encrypted log file.</li>
<li><code>-out decrypted_logs.txt</code>decrypted logs will be saved into this file.</li>
</ul>
<ol start="2">
<li>View the decrypted logs
Once decrypted, you can view the logs by simply running:</li>
</ol>
<pre><code class="language-bash">cat decrypted_logs.txt
</code></pre>
<p>This command will output the decrypted logs to the terminal.</p>
