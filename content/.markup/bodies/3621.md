<pre><code class="language-bash">curl -X 'POST' 'https://api.cloudflare.com/client/v4/accounts/&lt;accountId&gt;/browser-rendering/markdown' \
  -H 'Content-Type: application/json' \
  -H 'Authorization: Bearer &lt;apiToken&gt;' \
  -d '{
    &quot;url&quot;: &quot;https://example.com&quot;
  }'
</code></pre>
