<pre><code class="language-typescript">import Cloudflare from &quot;cloudflare&quot;;

const client = new Cloudflare({
	apiToken: process.env.API_TOKEN,
});

async function deployUserWorker(
	accountId: string,
	namespace: string,
	scriptName: string,
	scriptContent: string,
) {
	const scriptFile = new File([scriptContent], `${scriptName}.mjs`, {
		type: &quot;application/javascript+module&quot;,
	});

	const result =
		await client.workersForPlatforms.dispatch.namespaces.scripts.update(
			namespace,
			scriptName,
			{
				account_id: accountId,
				metadata: {
					main_module: `${scriptName}.mjs`,
				},
				files: [scriptFile],
			},
		);

	return result;
}

// Usage
await deployUserWorker(
	&quot;your-account-id&quot;,
	&quot;production&quot;,
	&quot;customer-123&quot;,
	`export default {
  async fetch(request, env, ctx) {
    return new Response(&quot;Hello from customer 123!&quot;);
  },
};`,
);
</code></pre>
