  account_id = var.cloudflare_account_id
  name       = &quot;Block requests from high-risk countries&quot;
  decision   = &quot;deny&quot;
  include = [{
    geo = {
      country_code = &quot;RU&quot;
    }
  }]
  exclude = [{
    ip_list = {
      id = cloudflare_zero_trust_list.corporate_ip_allowlist.id
    }
  }]
}
</code></pre>
