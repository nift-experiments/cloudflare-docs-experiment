<p>Use <a href="https://www.npmjs.com/package/postal-mime"><code>postal-mime</code></a> to parse the MIME structure of an incoming email. The parser handles multipart boundaries, transfer encodings, and character sets correctly.</p>
<pre><code class="language-ts">import PostalMime from &quot;postal-mime&quot;;

export default {
	async email(message, env, ctx): Promise&lt;void&gt; {
		const email = await PostalMime.parse(message.raw);

		console.log(`Subject: ${email.subject}`);
		console.log(`Text: ${email.text}`);
		console.log(`HTML: ${email.html}`);
	},
};
