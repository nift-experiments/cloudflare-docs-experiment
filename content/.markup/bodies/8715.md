<section class="nb-tab-panel" role="tabpanel" data-nb-tabs-content data-nb-tab-label="With binding">

<p>Pass the Flagship binding directly to the provider. This is the recommended approach inside a Worker.</p>
<div class="nb-type-script-example">
<pre><code class="language-ts">import { OpenFeature } from &quot;@openfeature/server-sdk&quot;;
import { FlagshipServerProvider } from &quot;@cloudflare/flagship/server&quot;;

export default {
	async fetch(request: Request, env: Env): Promise&lt;Response&gt; {
		await OpenFeature.setProviderAndWait(
			new FlagshipServerProvider({ binding: env.FLAGS }),
		);

		const client = OpenFeature.getClient();

		const showNewCheckout = await client.getBooleanValue(
			&quot;new-checkout&quot;,
			false,
			{ targetingKey: &quot;user-42&quot;, plan: &quot;enterprise&quot; },
		);

		if (showNewCheckout) {
			return new Response(&quot;New checkout enabled!&quot;);
		}

		return new Response(&quot;Standard checkout.&quot;);
	},
};
</code></pre>
</div>

<p>Use an app ID, account ID, and an API token when running outside of a Worker (for example, in Node.js). Generate an <a href="/flagship/api-tokens/">API token</a> from your Cloudflare account with Flagship Evaluate or Flagship App Evaluate permission.</p>
<div class="nb-type-script-example">
<pre><code class="language-ts">import { OpenFeature } from &quot;@openfeature/server-sdk&quot;;
import { FlagshipServerProvider } from &quot;@cloudflare/flagship/server&quot;;

await OpenFeature.setProviderAndWait(
	new FlagshipServerProvider({
		appId: &quot;&lt;APP_ID&gt;&quot;,
		accountId: &quot;&lt;ACCOUNT_ID&gt;&quot;,
		authToken: &quot;&lt;API_TOKEN&gt;&quot;,
	}),
);

const client = OpenFeature.getClient();

const showNewCheckout = await client.getBooleanValue(&quot;new-checkout&quot;, false, {
	targetingKey: &quot;user-42&quot;,
	plan: &quot;enterprise&quot;,
});
</code></pre>
</div>
