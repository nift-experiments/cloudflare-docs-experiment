        hd = self.env.HYPERDRIVE
        connection = await aiomysql.connect(
            host=hd.host,
            port=int(hd.port),
            user=hd.user,
            password=hd.password,
            db=hd.database,
            ssl=None,
        )
        try:
            cursor = await connection.cursor()
            await cursor.execute(&quot;SELECT 1&quot;)
            result = await cursor.fetchone()
            return Response.json({&quot;result&quot;: result[0]})
        finally:
            connection.close()
