<p>After you have set up your database, you will run a SQL query from within your Worker.</p>
<p>Go to your <code>hyperdrive-tutorial</code> Worker and open the <code>index.ts</code> file.</p>
<p>The <code>index.ts</code> file is where you configure your Worker's interactions with Hyperdrive.</p>
<p>Populate your <code>index.ts</code> file with the following code:</p>
<pre><code class="language-typescript">// pg 8.13.0 or later is recommended
import { Client } from &quot;pg&quot;;

export interface Env {
	// If you set another name in the Wrangler config file as the value for 'binding',
	// replace &quot;HYPERDRIVE&quot; with the variable name you defined.
	HYPERDRIVE: Hyperdrive;
}

export default {
	async fetch(request, env, ctx): Promise&lt;Response&gt; {
		// Create a new client on each request. Hyperdrive maintains the underlying
		// database connection pool, so creating a new client is fast.
		const sql = new Client({
			connectionString: env.HYPERDRIVE.connectionString,
		});

		try {
			// Connect to the database
			await sql.connect();

			// Sample query
			const results = await sql.query(`SELECT * FROM pg_tables`);

			// Return result rows as JSON
			return Response.json(results.rows);
		} catch (e) {
			console.error(e);
			return Response.json(
				{ error: e instanceof Error ? e.message : e },
				{ status: 500 },
			);
		}
	},
} satisfies ExportedHandler&lt;Env&gt;;
</code></pre>
<p>Upon receiving a request, the code above does the following:</p>
<ol>
<li>Creates a new database client configured to connect to your database via Hyperdrive, using the Hyperdrive connection string.</li>
<li>Initiates a query via <code>await sql.query()</code> that outputs all tables (user and system created) in the database (as an example query).</li>
<li>Returns the response as JSON to the client. Hyperdrive automatically cleans up the client connection when the request ends, and keeps the underlying database connection open in its pool for reuse.</li>
</ol>

import { createConnection } from &quot;mysql2/promise&quot;;

export interface Env {
	// If you set another name in the Wrangler config file as the value for 'binding',
	// replace &quot;HYPERDRIVE&quot; with the variable name you defined.

export default {
	async fetch(request, env, ctx): Promise&lt;Response&gt; {
		// Create a new connection on each request. Hyperdrive maintains the underlying
		// database connection pool, so creating a new connection is fast.
		const connection = await createConnection({
			host: env.HYPERDRIVE.host,
			user: env.HYPERDRIVE.user,
			password: env.HYPERDRIVE.password,
			database: env.HYPERDRIVE.database,
			port: env.HYPERDRIVE.port,

			// The following line is needed for mysql2 compatibility with Workers
			// mysql2 uses eval() to optimize result parsing for rows with &gt; 100 columns
			// Configure mysql2 to use static parsing instead of eval() parsing with disableEval
			disableEval: true,
		});

		try {
			// Sample query
			const [results, fields] = await connection.query(&quot;SHOW tables;&quot;);

			// Return result rows as JSON
			return new Response(JSON.stringify({ results, fields }), {
				headers: {
					&quot;Content-Type&quot;: &quot;application/json&quot;,
					&quot;Access-Control-Allow-Origin&quot;: &quot;*&quot;,
				},
			});
		} catch (e) {
			console.error(e);
			return Response.json(
				{ error: e instanceof Error ? e.message : e },
				{ status: 500 },
			);
		}
	},
} satisfies ExportedHandler&lt;Env&gt;;
