--header &quot;Authorization: Bearer $CLOUDFLARE_API_TOKEN&quot; \
--header &quot;Content-Type: application/json&quot; \
--data '{
  &quot;name&quot;: &quot;Company Wiki DNS policy&quot;,
  &quot;conditions&quot;: [
    {
      &quot;type&quot;: &quot;traffic&quot;,
      &quot;expression&quot;: {
        &quot;any&quot;: {
          &quot;in&quot;: {
            &quot;lhs&quot;: {
              &quot;splat&quot;: &quot;dns.domains&quot;
            },
            &quot;rhs&quot;: &quot;$&lt;DOMAIN_LIST_ID&gt;&quot;
          }
        }
      }
    },
    {
      &quot;type&quot;: &quot;identity&quot;,
      &quot;expression&quot;: {
        &quot;matches&quot;: {
          &quot;lhs&quot;: &quot;identity.email&quot;,
          &quot;rhs&quot;: &quot;.*@example.com&quot;
        }
      }
    }
  ],
  &quot;action&quot;: &quot;allow&quot;,
  &quot;precedence&quot;: 13002,
  &quot;enabled&quot;: true,
  &quot;description&quot;: &quot;Allow employees to access company wiki domains.&quot;,
  &quot;filters&quot;: [
    &quot;dns&quot;
  ]
