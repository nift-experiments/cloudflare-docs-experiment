<p>This page lists and defines common patterns for implementing AI agents, based on <a href="https://www.anthropic.com/research/building-effective-agents">Anthropic's patterns for building effective agents</a>.</p>
<p>Code samples use the <a href="https://ai-sdk.dev/docs/foundations/agents">AI SDK</a>, running in <a href="/durable-objects">Durable Objects</a>.</p>
<h2 id="prompt-chaining">Prompt Chaining</h2>
<p>Decomposes tasks into a sequence of steps, where each LLM call processes the output of the previous one.</p>
<p><img src="/assets/upstream/images/agents/01-prompt-chaining.jpg" alt="Figure 1: Prompt Chaining" /></p>
<pre><code class="language-ts">import { openai } from &quot;@ai-sdk/openai&quot;;&#10;import { generateText, generateObject } from &quot;ai&quot;;&#10;import { z } from &quot;zod&quot;;&#10;&#10;export default async function generateMarketingCopy(input: string) {&#10;  const model = openai(&quot;gpt-4o&quot;);&#10;&#10;  // First step: Generate marketing copy&#10;  const { text: copy } = await generateText({&#10;    model,&#10;    prompt: `Write persuasive marketing copy for: ${input}. Focus on benefits and emotional appeal.`,&#10;  });&#10;&#10;  // Perform quality check on copy&#10;  const { object: qualityMetrics } = await generateObject({&#10;    model,&#10;    schema: z.object({&#10;      hasCallToAction: z.boolean(),&#10;      emotionalAppeal: z.number().min(1).max(10),&#10;      clarity: z.number().min(1).max(10),&#10;    }),&#10;    prompt: `Evaluate this marketing copy for:&#10;    1. Presence of call to action (true/false)&#10;    2. Emotional appeal (1-10)&#10;    3. Clarity (1-10)&#10;&#10;    Copy to evaluate: ${copy}`,&#10;  });&#10;&#10;  // If quality check fails, regenerate with more specific instructions&#10;  if (&#10;    !qualityMetrics.hasCallToAction ||&#10;    qualityMetrics.emotionalAppeal &lt; 7 ||&#10;    qualityMetrics.clarity &lt; 7&#10;  ) {&#10;    const { text: improvedCopy } = await generateText({&#10;      model,&#10;      prompt: `Rewrite this marketing copy with:&#10;      ${!qualityMetrics.hasCallToAction ? &quot;- A clear call to action&quot; : &quot;&quot;}&#10;      ${qualityMetrics.emotionalAppeal &lt; 7 ? &quot;- Stronger emotional appeal&quot; : &quot;&quot;}&#10;      ${qualityMetrics.clarity &lt; 7 ? &quot;- Improved clarity and directness&quot; : &quot;&quot;}&#10;&#10;      Original copy: ${copy}`,&#10;    });&#10;    return { copy: improvedCopy, qualityMetrics };&#10;  }&#10;&#10;  return { copy, qualityMetrics };&#10;}&#10;</code></pre>
<h2 id="routing">Routing</h2>
<p>Classifies input and directs it to specialized followup tasks, allowing for separation of concerns.</p>
<p><img src="/assets/upstream/images/agents/2_Routing.jpg" alt="Figure 2: Routing" /></p>
<pre><code class="language-ts">import { openai } from &#x27;@ai-sdk/openai&#x27;;&#10;import { generateObject, generateText } from &#x27;ai&#x27;;&#10;import { z } from &#x27;zod&#x27;;&#10;&#10;async function handleCustomerQuery(query: string) {&#10;  const model = openai(&#x27;gpt-4o&#x27;);&#10;&#10;  // First step: Classify the query type&#10;  const { object: classification } = await generateObject({&#10;    model,&#10;    schema: z.object({&#10;      reasoning: z.string(),&#10;      type: z.enum([&#x27;general&#x27;, &#x27;refund&#x27;, &#x27;technical&#x27;]),&#10;      complexity: z.enum([&#x27;simple&#x27;, &#x27;complex&#x27;]),&#10;    }),&#10;    prompt: `Classify this customer query:&#10;    ${query}&#10;&#10;    Determine:&#10;    1. Query type (general, refund, or technical)&#10;    2. Complexity (simple or complex)&#10;    3. Brief reasoning for classification`,&#10;  });&#10;&#10;  // Route based on classification&#10;  // Set model and system prompt based on query type and complexity&#10;  const { text: response } = await generateText({&#10;    model:&#10;      classification.complexity === &#x27;simple&#x27;&#10;        ? openai(&#x27;gpt-4o-mini&#x27;)&#10;        : openai(&#x27;o1-mini&#x27;),&#10;    system: {&#10;      general:&#10;        &#x27;You are an expert customer service agent handling general inquiries.&#x27;,&#10;      refund:&#10;        &#x27;You are a customer service agent specializing in refund requests. Follow company policy and collect necessary information.&#x27;,&#10;      technical:&#10;        &#x27;You are a technical support specialist with deep product knowledge. Focus on clear step-by-step troubleshooting.&#x27;,&#10;    }[classification.type],&#10;    prompt: query,&#10;  });&#10;&#10;  return { response, classification };&#10;}&#10;</code></pre>
<h2 id="parallelization">Parallelization</h2>
<p>Enables simultaneous task processing through sectioning or voting mechanisms.</p>
<p><img src="/assets/upstream/images/agents/3_Parallelization.jpg" alt="Figure 3: Parallelization" /></p>
<pre><code class="language-ts">import { openai } from &#x27;@ai-sdk/openai&#x27;;&#10;import { generateText, generateObject } from &#x27;ai&#x27;;&#10;import { z } from &#x27;zod&#x27;;&#10;&#10;// Example: Parallel code review with multiple specialized reviewers&#10;async function parallelCodeReview(code: string) {&#10;  const model = openai(&#x27;gpt-4o&#x27;);&#10;&#10;  // Run parallel reviews&#10;  const [securityReview, performanceReview, maintainabilityReview] =&#10;    await Promise.all([&#10;      generateObject({&#10;        model,&#10;        system:&#10;          &#x27;You are an expert in code security. Focus on identifying security vulnerabilities, injection risks, and authentication issues.&#x27;,&#10;        schema: z.object({&#10;          vulnerabilities: z.array(z.string()),&#10;          riskLevel: z.enum([&#x27;low&#x27;, &#x27;medium&#x27;, &#x27;high&#x27;]),&#10;          suggestions: z.array(z.string()),&#10;        }),&#10;        prompt: `Review this code:&#10;      ${code}`,&#10;      }),&#10;&#10;      generateObject({&#10;        model,&#10;        system:&#10;          &#x27;You are an expert in code performance. Focus on identifying performance bottlenecks, memory leaks, and optimization opportunities.&#x27;,&#10;        schema: z.object({&#10;          issues: z.array(z.string()),&#10;          impact: z.enum([&#x27;low&#x27;, &#x27;medium&#x27;, &#x27;high&#x27;]),&#10;          optimizations: z.array(z.string()),&#10;        }),&#10;        prompt: `Review this code:&#10;      ${code}`,&#10;      }),&#10;&#10;      generateObject({&#10;        model,&#10;        system:&#10;          &#x27;You are an expert in code quality. Focus on code structure, readability, and adherence to best practices.&#x27;,&#10;        schema: z.object({&#10;          concerns: z.array(z.string()),&#10;          qualityScore: z.number().min(1).max(10),&#10;          recommendations: z.array(z.string()),&#10;        }),&#10;        prompt: `Review this code:&#10;      ${code}`,&#10;      }),&#10;    ]);&#10;&#10;  const reviews = [&#10;    { ...securityReview.object, type: &#x27;security&#x27; },&#10;    { ...performanceReview.object, type: &#x27;performance&#x27; },&#10;    { ...maintainabilityReview.object, type: &#x27;maintainability&#x27; },&#10;  ];&#10;&#10;  // Aggregate results using another model instance&#10;  const { text: summary } = await generateText({&#10;    model,&#10;    system: &#x27;You are a technical lead summarizing multiple code reviews.&#x27;,&#10;    prompt: `Synthesize these code review results into a concise summary with key actions:&#10;    ${JSON.stringify(reviews, null, 2)}`,&#10;  });&#10;&#10;  return { reviews, summary };&#10;}&#10;</code></pre>
<h2 id="orchestrator-workers">Orchestrator-Workers</h2>
<p>A central LLM dynamically breaks down tasks, delegates to Worker LLMs, and synthesizes results.</p>
<p><img src="/assets/upstream/images/agents/4_Orchestrator-Workers.jpg" alt="Figure 4: Orchestrator Workers" /></p>
<pre><code class="language-ts">import { openai } from &#x27;@ai-sdk/openai&#x27;;&#10;import { generateObject } from &#x27;ai&#x27;;&#10;import { z } from &#x27;zod&#x27;;&#10;&#10;async function implementFeature(featureRequest: string) {&#10;  // Orchestrator: Plan the implementation&#10;  const { object: implementationPlan } = await generateObject({&#10;    model: openai(&#x27;o1&#x27;),&#10;    schema: z.object({&#10;      files: z.array(&#10;        z.object({&#10;          purpose: z.string(),&#10;          filePath: z.string(),&#10;          changeType: z.enum([&#x27;create&#x27;, &#x27;modify&#x27;, &#x27;delete&#x27;]),&#10;        }),&#10;      ),&#10;      estimatedComplexity: z.enum([&#x27;low&#x27;, &#x27;medium&#x27;, &#x27;high&#x27;]),&#10;    }),&#10;    system:&#10;      &#x27;You are a senior software architect planning feature implementations.&#x27;,&#10;    prompt: `Analyze this feature request and create an implementation plan:&#10;    ${featureRequest}`,&#10;  });&#10;&#10;  // Workers: Execute the planned changes&#10;  const fileChanges = await Promise.all(&#10;    implementationPlan.files.map(async file =&gt; {&#10;      // Each worker is specialized for the type of change&#10;      const workerSystemPrompt = {&#10;        create:&#10;          &#x27;You are an expert at implementing new files following best practices and project patterns.&#x27;,&#10;        modify:&#10;          &#x27;You are an expert at modifying existing code while maintaining consistency and avoiding regressions.&#x27;,&#10;        delete:&#10;          &#x27;You are an expert at safely removing code while ensuring no breaking changes.&#x27;,&#10;      }[file.changeType];&#10;&#10;      const { object: change } = await generateObject({&#10;        model: openai(&#x27;gpt-4o&#x27;),&#10;        schema: z.object({&#10;          explanation: z.string(),&#10;          code: z.string(),&#10;        }),&#10;        system: workerSystemPrompt,&#10;        prompt: `Implement the changes for ${file.filePath} to support:&#10;        ${file.purpose}&#10;&#10;        Consider the overall feature context:&#10;        ${featureRequest}`,&#10;      });&#10;&#10;      return {&#10;        file,&#10;        implementation: change,&#10;      };&#10;    }),&#10;  );&#10;&#10;  return {&#10;    plan: implementationPlan,&#10;    changes: fileChanges,&#10;  };&#10;}&#10;</code></pre>
<h2 id="evaluator-optimizer">Evaluator-Optimizer</h2>
<p>One LLM generates responses while another provides evaluation and feedback in a loop.</p>
<p><img src="/assets/upstream/images/agents/5_Evaluator-Optimizer.jpg" alt="Figure 5: Evaluator-Optimizer" /></p>
<pre><code class="language-ts">import { openai } from &#x27;@ai-sdk/openai&#x27;;&#10;import { generateText, generateObject } from &#x27;ai&#x27;;&#10;import { z } from &#x27;zod&#x27;;&#10;&#10;async function translateWithFeedback(text: string, targetLanguage: string) {&#10;  let currentTranslation = &#x27;&#x27;;&#10;  let iterations = 0;&#10;  const MAX_ITERATIONS = 3;&#10;&#10;  // Initial translation&#10;  const { text: translation } = await generateText({&#10;    model: openai(&#x27;gpt-4o-mini&#x27;), // use small model for first attempt&#10;    system: &#x27;You are an expert literary translator.&#x27;,&#10;    prompt: `Translate this text to ${targetLanguage}, preserving tone and cultural nuances:&#10;    ${text}`,&#10;  });&#10;&#10;  currentTranslation = translation;&#10;&#10;  // Evaluation-optimization loop&#10;  while (iterations &lt; MAX_ITERATIONS) {&#10;    // Evaluate current translation&#10;    const { object: evaluation } = await generateObject({&#10;      model: openai(&#x27;gpt-4o&#x27;), // use a larger model to evaluate&#10;      schema: z.object({&#10;        qualityScore: z.number().min(1).max(10),&#10;        preservesTone: z.boolean(),&#10;        preservesNuance: z.boolean(),&#10;        culturallyAccurate: z.boolean(),&#10;        specificIssues: z.array(z.string()),&#10;        improvementSuggestions: z.array(z.string()),&#10;      }),&#10;      system: &#x27;You are an expert in evaluating literary translations.&#x27;,&#10;      prompt: `Evaluate this translation:&#10;&#10;      Original: ${text}&#10;      Translation: ${currentTranslation}&#10;&#10;      Consider:&#10;      1. Overall quality&#10;      2. Preservation of tone&#10;      3. Preservation of nuance&#10;      4. Cultural accuracy`,&#10;    });&#10;&#10;    // Check if quality meets threshold&#10;    if (&#10;      evaluation.qualityScore &gt;= 8 &amp;&amp;&#10;      evaluation.preservesTone &amp;&amp;&#10;      evaluation.preservesNuance &amp;&amp;&#10;      evaluation.culturallyAccurate&#10;    ) {&#10;      break;&#10;    }&#10;&#10;    // Generate improved translation based on feedback&#10;    const { text: improvedTranslation } = await generateText({&#10;      model: openai(&#x27;gpt-4o&#x27;), // use a larger model&#10;      system: &#x27;You are an expert literary translator.&#x27;,&#10;      prompt: `Improve this translation based on the following feedback:&#10;      ${evaluation.specificIssues.join(&#x27;\n&#x27;)}&#10;      ${evaluation.improvementSuggestions.join(&#x27;\n&#x27;)}&#10;&#10;      Original: ${text}&#10;      Current Translation: ${currentTranslation}`,&#10;    });&#10;&#10;    currentTranslation = improvedTranslation;&#10;    iterations++;&#10;  }&#10;&#10;  return {&#10;    finalTranslation: currentTranslation,&#10;    iterationsRequired: iterations,&#10;  };&#10;}&#10;</code></pre>
