<p>Some AI providers support real-time, low-latency interactions over WebSockets. AI Gateway allows seamless integration with these APIs, supporting multimodal interactions such as text, audio, and video.</p>
<h2 id="supported-providers">Supported Providers</h2>
<ul>
<li><a href="https://platform.openai.com/docs/guides/realtime-websocket">OpenAI</a></li>
<li><a href="https://ai.google.dev/gemini-api/docs/multimodal-live">Google AI Studio</a></li>
<li><a href="https://docs.cartesia.ai/api-reference/tts/tts">Cartesia</a></li>
<li><a href="https://elevenlabs.io/docs/conversational-ai/api-reference/conversational-ai/websocket">ElevenLabs</a></li>
<li><a href="https://docs.fal.ai/model-apis/model-endpoints/websockets">Fal AI</a></li>
<li><a href="https://developers.cloudflare.com/workers-ai/models/?authors=deepgram">Deepgram (Workers AI)</a></li>
</ul>
<h2 id="authentication">Authentication</h2>
<p>For real-time WebSockets, authentication can be done using:</p>
<ul>
<li>Headers (for non-browser environments)</li>
<li><code>sec-websocket-protocol</code> (for browsers)</li>
</ul>

<h3 id="deepgram-workers-ai">Deepgram (Workers AI)</h3>
<p>Workers AI provides Deepgram models for real-time speech-to-text (STT) and text-to-speech (TTS) capabilities through WebSocket connections.</p>
<h4 id="speech-to-text-stt">Speech-to-Text (STT)</h4>
<p>Workers AI supports two Deepgram STT models: <code>@cf/deepgram/nova-3</code> and <code>@cf/deepgram/flux</code>. The following example demonstrates real-time audio transcription from a microphone:</p>
<pre tabindex="0"><code class="language-javascript">import WebSocket from &quot;ws&quot;;&#10;import mic from &quot;mic&quot;;&#10;&#10;const ws = new WebSocket(&#10;	&quot;wss://gateway.ai.cloudflare.com/v1/&lt;account_id&gt;/&lt;gateway&gt;/workers-ai?model=@cf/deepgram/nova-3&amp;encoding=linear16&amp;sample_rate=16000&amp;interim_results=true&quot;,&#10;	{&#10;		headers: {&#10;			&quot;cf-aig-authorization&quot;: process.env.CLOUDFLARE_API_KEY,&#10;		},&#10;	},&#10;);&#10;&#10;// Configure microphone&#10;const micInstance = mic({&#10;	rate: &quot;16000&quot;,&#10;	channels: &quot;1&quot;,&#10;	debug: false,&#10;	exitOnSilence: 6,&#10;});&#10;&#10;const micInputStream = micInstance.getAudioStream();&#10;&#10;micInputStream.on(&quot;data&quot;, (data) =&gt; {&#10;	if (ws.readyState === WebSocket.OPEN) {&#10;		ws.send(data);&#10;	}&#10;});&#10;&#10;micInputStream.on(&quot;error&quot;, (error) =&gt; {&#10;	console.error(&quot;Microphone error:&quot;, error);&#10;});&#10;&#10;ws.onopen = () =&gt; {&#10;	console.log(&quot;Connected to WebSocket&quot;);&#10;	console.log(&quot;Starting microphone...&quot;);&#10;	micInstance.start();&#10;};&#10;&#10;ws.onmessage = (event) =&gt; {&#10;	try {&#10;		const parse = JSON.parse(event.data);&#10;		if (parse.channel?.alternatives?.[0]?.transcript) {&#10;			if (parse.is_final) {&#10;				console.log(&#10;					&quot;Final transcript:&quot;,&#10;					parse.channel.alternatives[0].transcript,&#10;				);&#10;			} else {&#10;				console.log(&#10;					&quot;Interim transcript:&quot;,&#10;					parse.channel.alternatives[0].transcript,&#10;				);&#10;			}&#10;		}&#10;	} catch (error) {&#10;		console.error(&quot;Error parsing message:&quot;, error);&#10;	}&#10;};&#10;&#10;ws.onerror = (error) =&gt; {&#10;	console.error(&quot;WebSocket error:&quot;, error);&#10;};&#10;&#10;ws.onclose = () =&gt; {&#10;	console.log(&quot;WebSocket closed&quot;);&#10;	micInstance.stop();&#10;};&#10;</code></pre>
<h4 id="text-to-speech-tts">Text-to-Speech (TTS)</h4>
<p>Workers AI supports the Deepgram <code>@cf/deepgram/aura-1</code> model for TTS. The following example demonstrates converting text input to audio:</p>
<pre tabindex="0"><code class="language-javascript">import WebSocket from &quot;ws&quot;;&#10;import readline from &quot;readline&quot;;&#10;import Speaker from &quot;speaker&quot;;&#10;&#10;const ws = new WebSocket(&#10;	&quot;wss://gateway.ai.cloudflare.com/v1/&lt;account_id&gt;/&lt;gateway&gt;/workers-ai?model=@cf/deepgram/aura-1&quot;,&#10;	{&#10;		headers: {&#10;			&quot;cf-aig-authorization&quot;: process.env.CLOUDFLARE_API_KEY,&#10;		},&#10;	},&#10;);&#10;&#10;// Speaker management&#10;let currentSpeaker = null;&#10;let isPlayingAudio = false;&#10;&#10;// Setup readline for text input&#10;const rl = readline.createInterface({&#10;	input: process.stdin,&#10;	output: process.stdout,&#10;	prompt: &quot;Enter text to speak (or \&quot;quit\&quot; to exit): &quot;,&#10;});&#10;&#10;ws.onopen = () =&gt; {&#10;	console.log(&quot;Connected to Deepgram TTS WebSocket&quot;);&#10;	rl.prompt();&#10;};&#10;&#10;ws.onmessage = (event) =&gt; {&#10;	// Check if message is JSON (metadata, flushed, etc.) or raw audio&#10;	if (event.data instanceof Buffer || event.data instanceof ArrayBuffer) {&#10;		// Raw audio data - create new speaker if needed&#10;		if (!currentSpeaker) {&#10;			currentSpeaker = new Speaker({&#10;				channels: 1,&#10;				bitDepth: 16,&#10;				sampleRate: 24000,&#10;			});&#10;			isPlayingAudio = true;&#10;		}&#10;		currentSpeaker.write(Buffer.from(event.data));&#10;	} else {&#10;		try {&#10;			const message = JSON.parse(event.data);&#10;			switch (message.type) {&#10;				case &quot;Metadata&quot;:&#10;					console.log(&quot;Model info:&quot;, message.model_name, message.model_version);&#10;					break;&#10;				case &quot;Flushed&quot;:&#10;					console.log(&quot;Audio complete&quot;);&#10;					// End speaker after flush to prevent buffer underflow&#10;					if (currentSpeaker &amp;&amp; isPlayingAudio) {&#10;						currentSpeaker.end();&#10;						currentSpeaker = null;&#10;						isPlayingAudio = false;&#10;					}&#10;					rl.prompt();&#10;					break;&#10;				case &quot;Cleared&quot;:&#10;					console.log(&quot;Audio cleared, sequence:&quot;, message.sequence_id);&#10;					break;&#10;				case &quot;Warning&quot;:&#10;					console.warn(&quot;Warning:&quot;, message.description);&#10;					break;&#10;			}&#10;		} catch (error) {&#10;			// Not JSON, might be raw audio as string&#10;			if (!currentSpeaker) {&#10;				currentSpeaker = new Speaker({&#10;					channels: 1,&#10;					bitDepth: 16,&#10;					sampleRate: 24000,&#10;				});&#10;				isPlayingAudio = true;&#10;			}&#10;			currentSpeaker.write(Buffer.from(event.data));&#10;		}&#10;	}&#10;};&#10;&#10;ws.onerror = (error) =&gt; {&#10;	console.error(&quot;WebSocket error:&quot;, error);&#10;};&#10;&#10;ws.onclose = () =&gt; {&#10;	console.log(&quot;WebSocket closed&quot;);&#10;	if (currentSpeaker) {&#10;		currentSpeaker.end();&#10;	}&#10;	rl.close();&#10;	process.exit(0);&#10;};&#10;&#10;// Handle user input&#10;rl.on(&quot;line&quot;, (input) =&gt; {&#10;	const text = input.trim();&#10;&#10;	if (text.toLowerCase() === &quot;quit&quot;) {&#10;		// Send Close message&#10;		ws.send(JSON.stringify({ type: &quot;Close&quot; }));&#10;		ws.close();&#10;		return;&#10;	}&#10;&#10;	if (text.length &gt; 0) {&#10;		// Send text to TTS&#10;		ws.send(&#10;			JSON.stringify({&#10;				type: &quot;Speak&quot;,&#10;				text: text,&#10;			}),&#10;		);&#10;&#10;		// Flush to get audio immediately&#10;		ws.send(JSON.stringify({ type: &quot;Flush&quot; }));&#10;		console.log(&quot;Flushing audio&quot;);&#10;	}&#10;&#10;	rl.prompt();&#10;});&#10;&#10;rl.on(&quot;close&quot;, () =&gt; {&#10;	if (ws.readyState === WebSocket.OPEN) {&#10;		ws.close();&#10;	}&#10;});&#10;</code></pre>
