<p class="article-summary">Execute Workers code in reaction to Container status changes</p>
<p>When a Container starts, stops, becomes idle, and errors, it can trigger code execution in a Worker
that has defined status hooks on the <code>Container</code> class. Refer to the <a href="/containers/reference/container-class/#lifecycle-hooks">Container class lifecycle hooks</a> for more details.</p>
<pre><code class="language-ts">import { Container } from &quot;@cloudflare/containers&quot;;&#10;&#10;export class MyContainer extends Container {&#10;	defaultPort = 4000;&#10;	sleepAfter = &quot;5m&quot;;&#10;&#10;	override onStart() {&#10;		console.log(&quot;Container successfully started&quot;);&#10;	}&#10;&#10;	override onStop(stopParams) {&#10;		if (stopParams.exitCode === 0) {&#10;			console.log(&quot;Container stopped gracefully&quot;);&#10;		} else {&#10;			console.log(&quot;Container stopped with exit code:&quot;, stopParams.exitCode);&#10;		}&#10;&#10;		console.log(&quot;Container stop reason:&quot;, stopParams.reason);&#10;	}&#10;&#10;	override async onActivityExpired() {&#10;		console.log(&quot;Container became idle, stopping it now&quot;);&#10;		await this.stop();&#10;	}&#10;&#10;	override onError(error: string) {&#10;		console.log(&quot;Container error:&quot;, error);&#10;	}&#10;}&#10;</code></pre>
