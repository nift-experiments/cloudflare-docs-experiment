<p>The Worker exposes a <code>POST /send-magic-link</code> route that validates the submitted email address, generates a single-use token, and emails the recipient a time-limited login link. The following code implements that handler.</p>
<pre><code class="language-typescript">interface Env {&#10;	EMAIL: SendEmail;&#10;	DOMAIN: string;&#10;}&#10;&#10;export default {&#10;	async fetch(request: Request, env: Env): Promise&lt;Response&gt; {&#10;		const url = new URL(request.url);&#10;&#10;		if (url.pathname === &quot;/send-magic-link&quot; &amp;&amp; request.method === &quot;POST&quot;) {&#10;			return handleSendMagicLink(request, env);&#10;		}&#10;&#10;		return new Response(&quot;Not Found&quot;, { status: 404 });&#10;	},&#10;};&#10;&#10;async function handleSendMagicLink(&#10;	request: Request,&#10;	env: Env,&#10;): Promise&lt;Response&gt; {&#10;	const { email } = await request.json();&#10;&#10;	if (!email || !isValidEmail(email)) {&#10;		return new Response(JSON.stringify({ error: &quot;Invalid email&quot; }), {&#10;			status: 400,&#10;		});&#10;	}&#10;&#10;	// Generate a simple secure token (you would implement proper JWT/token handling)&#10;	const token = crypto.randomUUID();&#10;	const magicUrl = `https://${env.DOMAIN}/login?token=${token}`;&#10;&#10;	// Send magic link email&#10;	await env.EMAIL.send({&#10;		to: email,&#10;		from: `noreply@${env.DOMAIN}`,&#10;		subject: &quot;Your login link&quot;,&#10;		html: `&#10;			&lt;h1&gt;Login to your account&lt;/h1&gt;&#10;			&lt;p&gt;Click the link below to log in:&lt;/p&gt;&#10;			&lt;p&gt;&lt;a href=&quot;${magicUrl}&quot;&gt;Login Now&lt;/a&gt;&lt;/p&gt;&#10;			&lt;p&gt;This link expires in 15 minutes.&lt;/p&gt;&#10;		`,&#10;		text: `&#10;			Login to your account&#10;			&#10;			Click this link to log in: ${magicUrl}&#10;			&#10;			This link expires in 15 minutes.&#10;		`,&#10;	});&#10;&#10;	return new Response(&#10;		JSON.stringify({&#10;			success: true,&#10;			message: &quot;Magic link sent to your email&quot;,&#10;		}),&#10;	);&#10;}&#10;&#10;function isValidEmail(email: string): boolean {&#10;	return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);&#10;}&#10;</code></pre>
