<p class="article-summary">A Pages Functions for appending CORS headers.</p>
<p>This example is a snippet from our Cloudflare Pages Template repo.</p>
<pre tabindex="0"><code class="language-ts">// Respond to OPTIONS method&#10;export const onRequestOptions: PagesFunction = async () =&gt; {&#10;	return new Response(null, {&#10;		status: 204,&#10;		headers: {&#10;			&quot;Access-Control-Allow-Origin&quot;: &quot;*&quot;,&#10;			&quot;Access-Control-Allow-Headers&quot;: &quot;*&quot;,&#10;			&quot;Access-Control-Allow-Methods&quot;: &quot;GET, OPTIONS&quot;,&#10;			&quot;Access-Control-Max-Age&quot;: &quot;86400&quot;,&#10;		},&#10;	});&#10;};&#10;&#10;// Set CORS to all /api responses&#10;export const onRequest: PagesFunction = async (context) =&gt; {&#10;	const response = await context.next();&#10;	response.headers.set(&quot;Access-Control-Allow-Origin&quot;, &quot;*&quot;);&#10;	response.headers.set(&quot;Access-Control-Max-Age&quot;, &quot;86400&quot;);&#10;	return response;&#10;};&#10;</code></pre>
