<p>You must <a href="/r2/api/tokens/">generate an Access Key</a> before getting started. All examples will utilize <code>access_key_id</code> and <code>access_key_secret</code> variables which represent the <strong>Access Key ID</strong> and <strong>Secret Access Key</strong> values you generated.</p>
<br />
<p><a href="https://www.npmjs.com/package/s3mini"><code>s3mini</code></a> is a zero-dependency, lightweight (~20 KB minified) TypeScript S3 client that uses AWS SigV4 signing. It runs natively on Node.js, Bun, and Cloudflare Workers without polyfills.</p>
<p>Unlike the AWS SDKs, s3mini expects a <strong>bucket-scoped endpoint</strong> — the bucket name is part of the endpoint URL, so you do not pass a separate <code>bucket</code> parameter to each operation.</p>
<h2 id="install">Install</h2>
<pre tabindex="0"><code class="language-sh">npm install s3mini&#10;</code></pre>
<h2 id="node-js-bun">Node.js / Bun</h2>
<pre tabindex="0"><code class="language-ts">import { S3mini } from &quot;s3mini&quot;;&#10;&#10;const s3 = new S3mini({&#10;	accessKeyId: process.env.R2_ACCESS_KEY_ID!,&#10;	secretAccessKey: process.env.R2_SECRET_ACCESS_KEY!,&#10;	// Bucket-scoped endpoint — include your bucket name in the path&#10;	endpoint: `https://${process.env.ACCOUNT_ID}.r2.cloudflarestorage.com/my-bucket`,&#10;	region: &quot;auto&quot;,&#10;});&#10;&#10;// Upload an object&#10;await s3.putObject(&quot;hello.txt&quot;, &quot;Hello from s3mini!&quot;);&#10;&#10;// Download an object as a string&#10;const text = await s3.getObject(&quot;hello.txt&quot;);&#10;console.log(text);&#10;&#10;// List objects with a prefix&#10;const objects = await s3.listObjects(&quot;/&quot;, &quot;hello&quot;);&#10;console.log(objects);&#10;&#10;// Delete an object&#10;await s3.deleteObject(&quot;hello.txt&quot;);&#10;</code></pre>

<pre tabindex="0"><code class="language-ts">import { S3mini } from &quot;s3mini&quot;;&#10;&#10;interface Env {&#10;	R2_ACCESS_KEY_ID: string;&#10;	R2_SECRET_ACCESS_KEY: string;&#10;	ACCOUNT_ID: string;&#10;}&#10;&#10;export default {&#10;	async fetch(request: Request, env: Env): Promise&lt;Response&gt; {&#10;		const s3 = new S3mini({&#10;			accessKeyId: env.R2_ACCESS_KEY_ID,&#10;			secretAccessKey: env.R2_SECRET_ACCESS_KEY,&#10;			endpoint: `https://${env.ACCOUNT_ID}.r2.cloudflarestorage.com/my-bucket`,&#10;			region: &quot;auto&quot;,&#10;		});&#10;&#10;		const url = new URL(request.url);&#10;		const key = url.pathname.slice(1); // strip leading &quot;/&quot;&#10;&#10;		if (!key) {&#10;			return new Response(&quot;Missing object key&quot;, { status: 400 });&#10;		}&#10;&#10;		switch (request.method) {&#10;			case &quot;PUT&quot;: {&#10;				const data = await request.arrayBuffer();&#10;				const contentType =&#10;					request.headers.get(&quot;content-type&quot;) ?? &quot;application/octet-stream&quot;;&#10;				await s3.putObject(key, new Uint8Array(data), contentType);&#10;				return new Response(&quot;Created&quot;, { status: 201 });&#10;			}&#10;&#10;			case &quot;GET&quot;: {&#10;				const response = await s3.getObjectResponse(key);&#10;				if (!response) {&#10;					return new Response(&quot;Not Found&quot;, { status: 404 });&#10;				}&#10;				return new Response(response.body, {&#10;					headers: {&#10;						&quot;content-type&quot;:&#10;							response.headers.get(&quot;content-type&quot;) ??&#10;							&quot;application/octet-stream&quot;,&#10;						etag: response.headers.get(&quot;etag&quot;) ?? &quot;&quot;,&#10;					},&#10;				});&#10;			}&#10;&#10;			case &quot;DELETE&quot;: {&#10;				await s3.deleteObject(key);&#10;				return new Response(null, { status: 204 });&#10;			}&#10;&#10;			default:&#10;				return new Response(&quot;Method Not Allowed&quot;, { status: 405 });&#10;		}&#10;	},&#10;};&#10;</code></pre>
