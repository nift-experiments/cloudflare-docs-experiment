<p class="article-summary">Set up an A/B test by controlling what response is served based on cookies.</p>
<p>This version passes through requests for <code>/test/*</code> and <code>/control/*</code> URI paths to the origin server, bypassing random assignment.</p>
<pre><code class="language-js">const NAME = &quot;myExampleABTest&quot;;&#10;&#10;export default {&#10;	async fetch(request) {&#10;		// Clone the original URL&#10;		const url = new URL(request.url);&#10;&#10;		// Enable Passthrough to allow direct access to control and test routes.&#10;		if (url.pathname.startsWith(&quot;/control&quot;) || url.pathname.startsWith(&quot;/test&quot;))&#10;			return fetch(request);&#10;&#10;		// Determine which group this requester is in.&#10;		const cookie = request.headers.get(&quot;cookie&quot;);&#10;&#10;		if (cookie &amp;&amp; cookie.includes(`${NAME}=control`)) {&#10;			url.pathname = &quot;/control&quot; + url.pathname;&#10;		} else if (cookie &amp;&amp; cookie.includes(`${NAME}=test`)) {&#10;			url.pathname = &quot;/test&quot; + url.pathname;&#10;		} else {&#10;			// If there is no cookie, this is a new client. Choose a group and set the cookie.&#10;			const group = Math.random() &lt; 0.5 ? &quot;test&quot; : &quot;control&quot;; // 50/50 split&#10;			if (group === &quot;control&quot;) {&#10;				url.pathname = &quot;/control&quot; + url.pathname;&#10;			} else {&#10;				url.pathname = &quot;/test&quot; + url.pathname;&#10;			}&#10;			// Reconstruct response to avoid immutability&#10;			let response = await fetch(url);&#10;			response = new Response(response.body, response);&#10;			// Set cookie to enable persistent A/B sessions.&#10;			response.headers.append(&quot;Set-Cookie&quot;, `${NAME}=${group}; path=/`);&#10;			return response;&#10;		}&#10;		return fetch(url);&#10;	},&#10;};&#10;</code></pre>
