<p>A very common use case is to provide the LLM with the ability to perform API calls via function calling.</p>
<p>In this example the LLM will retrieve the weather forecast for the next 5 days.
To do so a <code>getWeather</code> function is defined that is passed to the LLM as tool.</p>
<p>The <code>getWeather</code>function extracts the user's location from the request and calls the external weather API via the Workers' <a href="/workers/runtime-apis/fetch/"><code>Fetch API</code></a> and returns the result.</p>
<pre tabindex="0"><code class="language-ts">import { runWithTools } from &#x27;@cloudflare/ai-utils&#x27;;&#10;&#10;type Env = {&#10;	AI: Ai;&#10;};&#10;&#10;export default {&#10;	async fetch(request, env, ctx) {&#10;		// Define function&#10;		const getWeather = async (args: { numDays: number }) =&gt; {&#10;			const { numDays } = args;&#10;      // Location is extracted from request based on&#10;      // https://developers.cloudflare.com/workers/runtime-apis/request/#incomingrequestcfproperties&#10;      const lat = request.cf?.latitude&#10;      const long = request.cf?.longitude&#10;&#10;      // Interpolate values for external API call&#10;			const response = await fetch(&#10;				`https://api.open-meteo.com/v1/forecast?latitude=${lat}&amp;longitude=${long}&amp;daily=temperature_2m_max,precipitation_sum&amp;timezone=GMT&amp;forecast_days=${numDays}`&#10;			);&#10;			return response.text();&#10;		};&#10;		// Run AI inference with function calling&#10;		const response = await runWithTools(&#10;			env.AI,&#10;			// Model with function calling support&#10;			&#x27;@hf/nousresearch/hermes-2-pro-mistral-7b&#x27;,&#10;			{&#10;				// Messages&#10;				messages: [&#10;					{&#10;						role: &#x27;user&#x27;,&#10;						content: &#x27;What the weather like the next 5 days? Respond as text&#x27;,&#10;					},&#10;				],&#10;				// Definition of available tools the AI model can leverage&#10;				tools: [&#10;					{&#10;						name: &#x27;getWeather&#x27;,&#10;						description: &#x27;Get the weather for the next [numDays] days&#x27;,&#10;						parameters: {&#10;							type: &#x27;object&#x27;,&#10;							properties: {&#10;								numDays: { type: &#x27;numDays&#x27;, description: &#x27;number of days for the weather forecast&#x27; },&#10;							},&#10;							required: [&#x27;numDays&#x27;],&#10;						},&#10;						// reference to previously defined function&#10;						function: getWeather,&#10;					},&#10;				],&#10;			}&#10;		);&#10;		return new Response(JSON.stringify(response));&#10;	},&#10;} satisfies ExportedHandler&lt;Env&gt;;&#10;</code></pre>
