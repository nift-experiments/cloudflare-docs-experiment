<p>Oftentimes APIs are defined and documented via <a href="https://swagger.io/specification/">OpenAPI specification</a>. The Cloudflare <code>ai-utils</code> package's <code>createToolsFromOpenAPISpec</code> function creates tools from the OpenAPI spec, which the LLM can then leverage to fulfill the prompt.</p>
<p>In this example the LLM will describe the a Github user, based Github's API and its OpenAPI spec.</p>
<pre tabindex="0"><code class="language-ts">import { createToolsFromOpenAPISpec, runWithTools } from &#x27;@cloudflare/ai-utils&#x27;;&#10;&#10;type Env = {&#10;	AI: Ai;&#10;};&#10;&#10;const APP_NAME = &#x27;cf-fn-calling-example-app&#x27;;&#10;&#10;export default {&#10;	async fetch(request, env, ctx) {&#10;		const toolsFromOpenAPISpec = [&#10;			// You can pass the OpenAPI spec link or contents directly&#10;			...(await createToolsFromOpenAPISpec(&#10;				&#x27;https://gist.githubusercontent.com/mchenco/fd8f20c8f06d50af40b94b0671273dc1/raw/f9d4b5cd5944cc32d6b34cad0406d96fd3acaca6/partial_api.github.com.json&#x27;,&#10;				{&#10;					overrides: [&#10;						{&#10;							matcher: ({ url }) =&gt; {&#10;								return url.hostname === &#x27;api.github.com&#x27;;&#10;							},&#10;							// for all requests on *.github.com, we&#x27;ll need to add a User-Agent.&#10;							values: {&#10;								headers: {&#10;									&#x27;User-Agent&#x27;: APP_NAME,&#10;								},&#10;							},&#10;						},&#10;					],&#10;				}&#10;			)),&#10;		];&#10;&#10;		const response = await runWithTools(&#10;			env.AI,&#10;			&#x27;@hf/nousresearch/hermes-2-pro-mistral-7b&#x27;,&#10;			{&#10;				messages: [&#10;					{&#10;						role: &#x27;user&#x27;,&#10;						content: &#x27;Who is cloudflare on Github and how many repos does the organization have?&#x27;,&#10;					},&#10;				],&#10;				tools: toolsFromOpenAPISpec,&#10;			}&#10;		);&#10;&#10;		return new Response(JSON.stringify(response));&#10;	},&#10;} satisfies ExportedHandler&lt;Env&gt;;&#10;</code></pre>
