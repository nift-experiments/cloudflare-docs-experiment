<p>The <a href="https://nodejs.org/api/stream.html">Node.js streams API</a> is the original API for working with streaming data in JavaScript, predating the <a href="https://streams.spec.whatwg.org/">WHATWG ReadableStream standard</a>. A stream is an abstract interface for working with streaming data in Node.js. Streams can be readable, writable, or both. All streams are instances of <a href="/workers/runtime-apis/nodejs/eventemitter/">EventEmitter</a>.</p>
<p>Where possible, you should use the <a href="https://streams.spec.whatwg.org/">WHATWG standard &quot;Web Streams&quot; API</a>, which is <a href="https://streams.spec.whatwg.org/">supported in Workers</a>.</p>
<pre><code class="language-js">import { Readable, Transform } from &quot;node:stream&quot;;&#10;&#10;import { text } from &quot;node:stream/consumers&quot;;&#10;&#10;import { pipeline } from &quot;node:stream/promises&quot;;&#10;&#10;// A Node.js-style Transform that converts data to uppercase&#10;// and appends a newline to the end of the output.&#10;class MyTransform extends Transform {&#10;	constructor() {&#10;		super({ encoding: &quot;utf8&quot; });&#10;	}&#10;	_transform(chunk, _, cb) {&#10;		this.push(chunk.toString().toUpperCase());&#10;		cb();&#10;	}&#10;	_flush(cb) {&#10;		this.push(&quot;\n&quot;);&#10;		cb();&#10;	}&#10;}&#10;&#10;export default {&#10;	async fetch() {&#10;		const chunks = [&#10;			&quot;hello &quot;,&#10;			&quot;from &quot;,&#10;			&quot;the &quot;,&#10;			&quot;wonderful &quot;,&#10;			&quot;world &quot;,&#10;			&quot;of &quot;,&#10;			&quot;node.js &quot;,&#10;			&quot;streams!&quot;,&#10;		];&#10;&#10;		function nextChunk(readable) {&#10;			readable.push(chunks.shift());&#10;			if (chunks.length === 0) readable.push(null);&#10;			else queueMicrotask(() =&gt; nextChunk(readable));&#10;		}&#10;&#10;		// A Node.js-style Readable that emits chunks from the&#10;		// array...&#10;		const readable = new Readable({&#10;			encoding: &quot;utf8&quot;,&#10;			read() {&#10;				nextChunk(readable);&#10;			},&#10;		});&#10;&#10;		const transform = new MyTransform();&#10;		await pipeline(readable, transform);&#10;		return new Response(await text(transform));&#10;	},&#10;};&#10;</code></pre>
